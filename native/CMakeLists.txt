cmake_minimum_required(VERSION 3.25)
project(xmp_dsp VERSION 0.1.0 LANGUAGES C CXX)
option(XMP_SANITIZE "Address/undefined-behavior instrumentation; separate from real-time audit" OFF)
if(XMP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Tests, command-line tools and their checks are built only when XMP OS is the
# top-level project. A program that adds XMP OS with add_subdirectory (such as
# xmp-modular, through its xmp-os submodule) gets the libraries below and the
# xmp:: targets, without XMP's own tests.
option(XMP_BUILD_TESTS "Build XMP OS's tests and tools" ${PROJECT_IS_TOP_LEVEL})

# The headers under include/xmp, for programs built on XMP's core. Most of the
# core (rack, sequencer, project, Live session pieces) is header-only.
add_library(xmp_headers INTERFACE)
target_include_directories(xmp_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(xmp_headers INTERFACE cxx_std_17)
add_library(xmp::headers ALIAS xmp_headers)
# Portable platform contracts shared with downstream native clients. Device
# resources remain owned by live_session; this target opens no device.
add_library(xmp_live_contracts INTERFACE)
target_link_libraries(xmp_live_contracts INTERFACE xmp_headers)
add_library(xmp::live_contracts ALIAS xmp_live_contracts)

add_library(xmp_dsp STATIC src/sampler.cpp src/gain.cpp)
target_include_directories(xmp_dsp PUBLIC include)
target_compile_options(xmp_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Threads REQUIRED)
add_library(xmp_rt_audit SHARED src/rt_audit.cpp)
target_include_directories(xmp_rt_audit PUBLIC include)
target_link_libraries(xmp_rt_audit PRIVATE Threads::Threads)
if(APPLE AND NOT XMP_SANITIZE)
  target_sources(xmp_rt_audit PRIVATE src/rt_interpose_mac.cpp)
endif()
add_library(xmp_engine_rt STATIC src/engine.cpp)
target_include_directories(xmp_engine_rt PUBLIC include)
target_compile_options(xmp_engine_rt PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(xmp_host_audio STATIC src/audio_backend.cpp)
target_include_directories(xmp_host_audio PUBLIC include)
target_link_libraries(xmp_host_audio PUBLIC xmp_engine_rt)
target_compile_options(xmp_host_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(APPLE)
  target_link_libraries(xmp_host_audio PUBLIC "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
elseif(UNIX)
  find_package(ALSA QUIET)
  if(ALSA_FOUND)
    target_compile_definitions(xmp_host_audio PRIVATE XMP_HAS_ALSA=1)
    target_link_libraries(xmp_host_audio PUBLIC ALSA::ALSA)
  endif()
endif()
add_library(xmp_parity_dsp STATIC src/parity_dsp.c)
target_include_directories(xmp_parity_dsp PUBLIC include/xmp)
target_compile_options(xmp_parity_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
if(UNIX AND NOT APPLE)
  target_link_libraries(xmp_parity_dsp PUBLIC m)
endif()

add_library(xmp::dsp ALIAS xmp_dsp)
add_library(xmp::rt_audit ALIAS xmp_rt_audit)
add_library(xmp::engine_rt ALIAS xmp_engine_rt)
add_library(xmp::host_audio ALIAS xmp_host_audio)
add_library(xmp::parity_dsp ALIAS xmp_parity_dsp)

if(XMP_BUILD_TESTS)
  include(cmake/xmp-tests.cmake)
endif()

add_library(xmp_live_platform STATIC src/live_platform.cpp)
target_link_libraries(xmp_live_platform PUBLIC xmp_headers)
target_compile_options(xmp_live_platform PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND ALSA_FOUND)
  target_sources(xmp_live_platform PRIVATE src/live_platform_linux.cpp)
  target_link_libraries(xmp_live_platform PUBLIC ALSA::ALSA)
endif()
add_library(xmp::live_platform ALIAS xmp_live_platform)
